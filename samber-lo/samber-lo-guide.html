samber/lo

基于 Go 1.18+ 泛型的 Lodash 风格工具库：200+ 帮助函数覆盖 slice、map、string、channel、并发与错误处理，把「写了几百遍的 for 循环样板」变成一行声明式代码。零第三方依赖，v1 严格遵循 SemVer。

github.com/samber/lo Go ≥ 1.18 零依赖 MIT License 文档 lo.samber.dev

1. 库简介

samber/lo 灵感来自 JavaScript 生态的 Lodash，只做一件事：把日常编码中重复出现的集合操作样板代码，封装成可复用的泛型函数。它不引入框架概念、不改变代码结构，即取即用。

// 没有 lo 的日子
names := make([]string, 0, len(users))
for _, u := range users {
    if u.Age >= 18 {
        names = append(names, u.Name)
    }
}

// 有 lo 的日子
names := lo.Map(lo.Filter(users, func(u User, _ int) bool {
    return u.Age >= 18
}), func(u User, _ int) string {
    return u.Name
})
go get github.com/samber/lo@v1
import (
    "github.com/samber/lo"
    lop "github.com/samber/lo/parallel" // 并行版本
    lom "github.com/samber/lo/mutable"  // 原地修改版本
)

同系列生态：mo（Monad：Option/Result）、do（依赖注入）、ro（响应式/无限流）。定位区分：lo 处理有限集合的同步计算，ro 处理无限事件流。

3. 设计原理

3.1 泛型签名：一套代码适配所有类型

3.2 Err 变体：函数式的错误传播

大部分转换函数都有 XxxErr 兄弟：回调多返回一个 error，遇到错误立即短路返回，把「循环里逐个 if err != nil」的模板也省掉了。

result, err := lo.MapErr([]int{1, 2, 3}, func(x int, _ int) (string, error) {
    if x == 3 {
        return "", fmt.Errorf("number 3 is not allowed")
    }
    return strconv.Itoa(x), nil
})
// result == nil, err == "number 3 is not allowed"

FilterErrMapErrFlatMapErrReduceErrReduceRightErrGroupByErr…全家族覆盖

3.3 子包矩阵

导入定位说明
lo同步 · 不可变主包，所有函数返回新集合，不修改入参
lo/parallel并发Map / ForEach 的并行版：每元素一个 goroutine，结果顺序与输入一致
lo/mutable原地修改复用底层数组、减少分配（注意副作用与数据竞争）
lo/it惰性迭代基于 Go 1.23 iter.Seq 的迭代器版本（实验性质）

3.4 性能画像

⚠️ Eager 求值：所有函数调用即完整遍历，中间产生临时切片。Map(Filter(xs)) 会遍历两次、分配两次 —— 可用 FilterMap 合并；极热路径上手写 for 循环仍最快（省去回调开销）。lo 的核心收益是可读性与正确性，不是极致性能。

4. Slice 实战：最常用的一组

转换与聚合

// Map：类型转换
lo.Map([]int64{1, 2, 3, 4}, func(x int64, _ int) string {
    return strconv.FormatInt(x, 10)
}) // []string{"1", "2", "3", "4"}

// FilterMap：过滤 + 转换一步到位（回调返回 (结果, 是否保留)）
lo.FilterMap([]string{"cpu", "gpu", "mouse"}, func(x string, _ int) (string, bool) {
    if strings.HasSuffix(x, "pu") {
        return "xpu", true
    }
    return "", false
}) // []string{"xpu", "xpu"}

// FlatMap：转换后拍平（返回 nil 则不加元素）
lo.FlatMap([]int64{0, 1, 2}, func(x int64, _ int) []string {
    return []string{strconv.FormatInt(x, 10), strconv.FormatInt(x, 10)}
}) // []string{"0","0","1","1","2","2"}

// Reduce：聚合（ReduceRight 反向）
lo.Reduce([]int{1, 2, 3, 4}, func(agg, item int, _ int) int { return agg + item }, 0) // 10

去重与分组

// Uniq / UniqMap / IsUniq
lo.Uniq([]int{1, 2, 2, 3})              // []int{1, 2, 3}
lo.UniqMap(users, func(u User, _ int) string { return u.Name })
// Map + 去重一步完成：[]string{"Alex", "Bob", "Alice"}

// GroupBy：按 key 分桶
lo.GroupBy([]int{0, 1, 2, 3, 4, 5}, func(item int, _ int) int {
    return item % 3
}) // map[int][]int{0: [0,3], 1: [1,4], 2: [2,5]}

// Chunk：切分固定大小子块（分批处理神器）
lo.Chunk([]int{0, 1, 2, 3, 4, 5}, 2) // [][]int{{0,1}, {2,3}, {4,5}}

// PartitionBy：按「连续」key 切段
lo.PartitionBy([]int{1, 1, 2, 2, 3}, func(x int, _ int) string {
    if x < 2 { return "low" } else { return "high" }
}) // [][]int{{1,1}, {2,2,3}}

// KeyBy / SliceToMap(Associate)：切片转 map
lo.SliceToMap(users, func(u User, _ int) (string, int) {
    return u.Name, u.Age
}) // map[string]int{"Alex": 10, ...}

5. Map / 查找 / 集合运算

//// Map 字典操作
lo.Keys(m) / lo.Values(m)              // 键/值切片
lo.PickBy(m, func(k string, v int) bool { return v > 10 }) // 保留满足项
lo.OmitBy(m, pred)                          // PickBy 反义
lo.Entries(map[string]int{"a": 1})    // []Tuple2{{"a",1}}
lo.FromEntries(pairs)                      // pairs → map（逆向）
lo.Invert(m)                                // 键值互换
lo.Assign(m1, m2, m3)                      // 合并（后者覆盖）
lo.MapKeys(m, fn) / lo.MapValues(m, fn)   // 变换键/值
lo.ValueOr(m, "k", 42)                    // 带默认值取值

//// 查找与判断
lo.IndexOf([]int{5, 6, 7}, 6)             // 1（不存在返回 -1）
v, ok := lo.Find(xs, pred)                 // (值, 是否找到)
lo.FindOrElse(xs, fallback, pred)           // 找不到给默认值
lo.FindKey(m, 20)                          // 按值反查键
lo.Contains(xs, 5)                         // 包含
lo.Every / lo.Some / lo.None               // 全部/任一/都不满足（均有 By 变体）

//// 集合运算
lo.Intersect(a, b)                          // 交集
lo.Difference(a, b)                        // 差集
lo.Union(a, b)                             // 并集
lo.Without(xs, 2, 3)                       // 剔除指定元素
lo.FindUniques / lo.FindDuplicates        // 只出现一次的 / 重复的

//// 极值与取样
lo.Min / lo.Max / lo.MinBy / lo.MaxBy / lo.MinIndexBy
lo.First / lo.Last / lo.Nth(xs, -1)            // 负索引支持
lo.FirstOr(xs, def)                         // 空集合安全取首
lo.Sample(xs) / lo.Samples(xs, 3)           // 随机取样

6. Tuple / 条件表达式 / 类型操作

Tuple：多返回值的容器化（最高 9 元）

// Zip：多个等长切片「拉链」成元组切片
lo.Zip2([]string{"a", "b"}, []int{1, 2})
// []lo.Tuple2[string,int]{{A:"a",B:1},{A:"b",B:2}}，最高 Zip9

// Unpack：元组解包成多返回值
a, b := lo.Unpack2(lo.Tuple2[string, int]{A: "x", B: 1})

// CrossJoin：笛卡尔积
lo.CrossJoin2([]string{"a"}, []int{1, 2})
// []Tuple2{{"a",1},{"a",2}}

条件表达式：让 Go 拥有「三元运算符」

// Ternary：注意两分支都会求值！
lo.Ternary(ok, "yes", "no")

// TernaryF：惰性求值版本（仅计算命中分支）
lo.TernaryF(ok, func() T { return expensive() }, func() T { return cheap() })

// 链式 if
lo.If(ok, "a").ElseIf(ok2, "b").Else("c")

// 链式 switch
lo.Switch(value).Case(1, "one").Case(2, "two").Default("many")

类型与指针：消灭 nil 判空样板

lo.ToPtr(42)               // *int：字面量直接取地址
lo.FromPtr(p)              // 解引用；nil → 零值
lo.FromPtrOr(p, 99)        // 解引用；nil → 默认值
lo.ToSlicePtr([]int{1, 2})   // []*int
lo.FromSlicePtrOr(ps, 0)   // []*int → []T（nil 安全）
lo.IsEmpty(xs) / lo.IsNotEmpty(xs)
lo.Coalesce(0, 1, 2)        // 返回第一个非零值 → 1（默认值串联神器）
lo.IsNil(x) / lo.IsNotNil(x)  // 任意类型的 nil 判断

// 字符串与数学（节选）
lo.RandomString(16, lo.LowerCaseLettersCharset)
lo.Substring("hello", 0, 3)    // "hel"（支持负索引）
lo.RuneLength("你好")          // 2（按 rune 而非 byte）
lo.SnakeCase("MyFunc")        // "my_func"
lo.CamelCase("http_server")   // "httpServer"
lo.KebabCase("HelloWorld")    // "hello-world"
lo.Range(5)                    // []int{0,1,2,3,4}
lo.RangeWithSteps(0, 10, 3)    // []int{0,3,6,9}
lo.Clamp(42, 0, 100)           // 区间钳制
lo.Sum / lo.Mean / lo.Mode       // 求和 / 均值 / 众数

7. 错误处理与并发控制

//// 错误处理
// Validate：条件不满足即返回错误（参数校验常用）
err := lo.Validate(user.Age >= 18, "user must be adult, got %d", user.Age)

// Must：「我保证不出错」，出错 panic（仅初始化期使用）
val := lo.Must(strconv.Atoi("42"))

// Try：函数式 recover
err = lo.Try(func() error { return doSomething() })
lo.TryCatch(fn, catchFn)                    // 捕获后交给 catch 处理

// ErrorsAs：一行完成 errors.As
if lo.ErrorsAs[*MyError](err) { ... }

//// 并发控制
// Attempt：带重试的执行（返回最终错误）
err = lo.Attempt(3, func(index int) error {
    return callFlakyAPI()
})

// AttemptWithDelay：带退避延迟的重试
err = lo.AttemptWithDelay(5, 2*time.Second, func(i int, d time.Duration) error {
    return retryableOp()
})

// Debounce：防抖（200ms 内多次调用只执行最后一次）
debounced := lo.Debounce(200*time.Millisecond, func() { search() })
debounced()

// Throttle：节流（1 秒内最多执行一次，其余丢弃）
throttled := lo.Throttle(1*time.Second, fn)

// Synchronize：把任意函数变成并发安全版本
safeFn := lo.Synchronize(fn)

// Async：立即起 goroutine，返回 channel 收结果（另有 Async0~Async6）
ch := lo.Async(func() int { return heavyCompute() })
result := <-ch

// WaitFor：轮询等待条件成立
err = lo.WaitFor(func() bool { return ready() }, 3*time.Second, 100*time.Millisecond)

// Transaction：多个函数顺序执行，遇错即停（并回滚你自行编排的清理逻辑）
err = lo.Transaction(fn1, fn2, fn3)

// 每 10 个元素或超时 1s 输出一批

10. 对比标准库 slices / maps

Go 1.21 起 slices / maps 进入标准库，功能有交集：

能力标准库samber/lo
Contains / Index / Max / Min✅ slices✅ 均有，另有 By / Err 变体
Clone / Sort / BinarySearch✅ slices（编译器深度优化）➖ 不做排序，保持不可变定位
Map / Filter / Reduce❌ 无✅ 核心优势
GroupBy / Chunk / KeyBy / Uniq❌ 无✅ 独有
Ternary / Coalesce / Attempt / Channel 系列❌ 无✅ 独有
✅ 共存策略：排序查找优先标准库（性能最优）；转换、分组、并发、错误处理类样板代码用 lo。两者可在同一项目共存，互不冲突。

11. 最佳实践与注意事项

🚫 超热路径慎用链式调用：Map(Filter(Map(xs))) 多次遍历 + 分配。用 FilterMap 合并，或回到手写 for。
⚠️ Ternary 两分支都会求值：含副作用或高成本计算时必须改用 TernaryF（惰性求值）。
⚠️ lo.Must 只用于初始化期：配置解析、正则编译等「出错即致命」的场景；业务路径禁用。
⚠️ parallel 回调安全性：回调被并发调用，闭包里的共享变量需自行加锁。
⚠️ 历史版本行为变化：lo.Reverse 等函数曾原地修改、新版改为返回新切片，升级版本前查 changelog，避免共享底层数组的坑。
⚠️ 区分子包别名：团队约定 lop / lom 别名，避免误用原地版本引发数据竞争。
✅ 可读性优先：lo 的价值是让代码意图（「我在做映射/过滤/分组」）一目了然。三层以上嵌套回调不如拆成命名函数 —— 一切以团队可维护性为准绳。
